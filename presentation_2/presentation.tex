\section{Przegląd zaimplementowanych modeli i ich architektur}

\begin{frame}{Zaimplementowane modele}
    \begin{block}{Podstawowe modele}
        \begin{itemize}
            \item \textbf{Autoencoder:} Klasyczny model do rekonstrukcji danych.
            \item \textbf{Conditional VAE:} Wariacyjny autoencoder z warstwą warunkującą, umożliwiający generowanie danych na podstawie etykiet.
            \item \textbf{Convolutional VAE:} Wariacyjny autoencoder z warstwami konwolucyjnymi, lepiej przystosowany do obrazów.
        \end{itemize}
    \end{block}
    \begin{block}{Zaawansowane modele}
        \begin{itemize}
            \item \textbf{Vector Quantized VAE (VQ-VAE):} Model łączący VAE z dyskretną przestrzenią latentną.
            \item \textbf{Generative Adversarial Network (GAN):} Model oparty na rywalizujących sieciach neuronowych.
            \item \textbf{Diffusion Model:} Model generatywny oparty na procesie redukcji szumu.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Porównanie modeli}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            \begin{itemize}
                \item \textbf{Autoencoder:}
                      \begin{itemize}
                          \item Najszybszy trening
                          \item Najprostsza implementacja
                          \item Ograniczona jakość generowanych próbek
                          \item BCE loss jako metryka rekonstrukcji
                      \end{itemize}
                \item \textbf{VAE:}
                      \begin{itemize}
                          \item Lepsza jakość generowanych próbek
                          \item Możliwość interpolacji w przestrzeni latentnej
                          \item Stabilny trening
                          \item Balans między rekonstrukcją a KL divergence
                      \end{itemize}
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \begin{itemize}
                \item \textbf{GAN:}
                      \begin{itemize}
                          \item Najlepsza jakość wizualna
                          \item Trudny trening
                          \item Problem z mode collapse
                          \item Wymaga balansu między generatorem a dyskryminatorem
                      \end{itemize}
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Zaawansowane modele}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Conditional VAE:}
            \begin{itemize}
                \item Kontrolowane generowanie cyfr
                \item Interpolacja między cyframi
                \item Lepsza jakość niż standardowy VAE
                \item Wymaga etykiet podczas treningu
            \end{itemize}

            \textbf{VQ-VAE:}
            \begin{itemize}
                \item Dyskretna przestrzeń latentna
                \item Wizualizacja codebooka
                \item Dobre kompresja z zachowaniem jakości
                \item Możliwość interpolacji
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Diffusion Model:}
            \begin{itemize}
                \item Progresywne generowanie
                \item Wysoka jakość próbek
                \item Długi czas generowania
                \item Stabilny proces treningu
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\section{Wnioski i kierunki rozwoju}

\begin{frame}{Główne wnioski}
    \begin{block}{Najważniejsze obserwacje}
        \begin{itemize}
            \item VAE i GAN osiągnęły najlepsze wyniki w generowaniu cyfr
            \item Conditional VAE oferuje najlepszy balans jakości i kontroli
            \item VQ-VAE zapewnia dobre kompresję z zachowaniem jakości
            \item Diffusion Model rozczarował - prawdopodobne błędy implementacyjne
        \end{itemize}
    \end{block}
    \begin{block}{Ranking modeli}
        \begin{enumerate}
            \item \textbf{Conditional VAE:} solidny wybór dla większości aplikacji.
            \item \textbf{Diffusion:} Najlepsza jakość, bardzo długi czas nauki i~generowania.
            \item \textbf{GAN:} Dobra jakość, trudny trening.
            \item \textbf{Convolutional VAE:} Lepsza jakość dla obrazów, wolniejszy trening.
            \item \textbf{VQ-VAE:} Dobra kompresja, problemy z jakością generowanych próbek.
            \item \textbf{Basic AE:} Najszybszy trening, ograniczona jakość.
        \end{enumerate}
    \end{block}
\end{frame}
