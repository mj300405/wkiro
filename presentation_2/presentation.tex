\section{Przegląd zaimplementowanych modeli i ich architektur}

\begin{frame}{Zaimplementowane modele}
    \begin{block}{Podstawowe modele}
        \begin{itemize}
            \item \textbf{Autoencoder:} Klasyczny model do rekonstrukcji danych.
            \item \textbf{Conditional Convolutional VAE:} Wariacyjny autoencoder z warstwą warunkującą, umożliwiający generowanie danych na podstawie etykiet.
            \item \textbf{Convolutional VAE:} Wariacyjny autoencoder z warstwami konwolucyjnymi, lepiej przystosowany do obrazów.
        \end{itemize}
    \end{block}
    \begin{block}{Zaawansowane modele}
        \begin{itemize}
            \item \textbf{Vector Quantized VAE (VQ-VAE):} Model łączący VAE z dyskretną przestrzenią latentną.
            \item \textbf{Generative Adversarial Network (GAN):} Model oparty na rywalizujących sieciach neuronowych.
            \item \textbf{Diffusion Model:} Model generatywny oparty na procesie redukcji szumu.
        \end{itemize}
    \end{block}
\end{frame}

\section{Wnioski}

\begin{frame}{Wnioski - podstawowe modele}
    \begin{columns}[T]
        \begin{column}{0.6\textwidth}
            \begin{itemize}
                \item \textbf{Autoencoder:}
                      \begin{itemize}
                          \item Najszybszy trening
                          \item Najprostsza implementacja
                          \item Ograniczona jakość generowanych próbek
                          \item BCE loss jako metryka rekonstrukcji
                      \end{itemize}
                \item \textbf{Convolutional VAE:}
                      \begin{itemize}
                          \item Lepsza jakość generowanych próbek
                          \item Możliwość interpolacji w przestrzeni latentnej
                          \item Stabilny trening
                          \item Balans między rekonstrukcją a KL divergence
                      \end{itemize}
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \begin{itemize}
                \item \textbf{GAN:}
                      \begin{itemize}
                          \item Trudny trening
                          \item Problem z mode collapse
                          \item Wymaga balansu między generatorem a dyskryminatorem
                      \end{itemize}
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Wnioski - zaawansowane modele}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{Conditional COnvolutional VAE:}
            \begin{itemize}
                \item Kontrolowane generowanie cyfr
                \item Interpolacja między cyframi
                \item Lepsza jakość niż standardowy VAE
                \item Wymaga etykiet podczas treningu
            \end{itemize}

            \textbf{VQ-VAE:}
            \begin{itemize}
                \item Dyskretna przestrzeń latentna
                \item Wizualizacja codebooka
                \item Dobre kompresja z zachowaniem jakości
                \item Możliwość interpolacji
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Diffusion Model:}
            \begin{itemize}
                \item Progresywne generowanie
                \item Wysoka jakość próbek
                \item Czyste, wyraźne cyfry
                \item Długi czas generowania
                \item Stabilny proces treningu
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Podsumowanie}
    \begin{block}{Najważniejsze obserwacje}
        \begin{itemize}
            \item Conditional Convolutional VAE i Diffusion Model osiągnęły najlepsze wyniki jakościowe.
            \item Conditional Convolutional VAE oferuje najlepszy balans jakości, szybkości i kontroli.
            \item Diffusion Model daje świetną jakość kosztem czasu generowania.
            \item GAN pozostaje dobrym wyborem mimo trudności w treningu.
            \item VQ-VAE zaskoczył szybkością konwergencji.
        \end{itemize}
    \end{block}

\end{frame}

\begin{frame}{Podsumowanie - cz. 2}
    \begin{block}{Ranking modeli}
        \begin{enumerate}
            \item \textbf{Conditional Convolutional VAE:} Najlepsza kontrola, doskonała jakość, łatwy w użyciu.
            \item \textbf{Diffusion:} Bardzo wysoka jakość, ale wolniejszy proces.
            \item \textbf{GAN:} Dobra jakość wizualna, trudniejszy trening.
            \item \textbf{Convolutional VAE:} Solidna jakość, dobry dla obrazów.
            \item \textbf{VQ-VAE:} Najszybszy trening, unikalna estetyka.
            \item \textbf{Basic AE:} Słaba jakość.
        \end{enumerate}
    \end{block}

    \begin{block}{Rekomendacje}
        \begin{itemize}
            \item Dla większości zastosowań: Conditional Convolutional VAE.
            \item Gdy jakość jest priorytetem: Diffusion Model.
        \end{itemize}
    \end{block}

\end{frame}
